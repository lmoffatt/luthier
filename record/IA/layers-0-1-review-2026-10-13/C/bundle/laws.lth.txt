import "checks.lth" Failures isEmpty

-- ---------------------------------------------------------------- §1 The verdict

-- A cost `C` (5.S2, 6.S5) with what a verdict needs of it: `zero`, holding exactly; `top`, failing;
-- and its order. The algebra of costs, with its six laws, is certification.lth's (step 5); here
-- only the two laws the retraction needs (5.S5): `zero <= zero`, and `top <= zero` false, so a
-- library cannot take `zero = top`.
CostScale (C : U) : { zero : C, top : C, leq : C -> C -> Bool }
ZeroIsExact (C : U) (s : CostScale C) : Bool := s.leq s.zero s.zero
TopIsNotExact (C : U) (s : CostScale C) : Bool := not (s.leq s.top s.zero)
CostScale (C : U) := ZeroIsExact C & TopIsNotExact C

-- `lift` reads a Boolean law as a cost, `exact` reads a cost back as whether it holds exactly
-- (5.S5); `exact (lift b) == b` follows from the two laws above
lift (C : U) (s : CostScale C) (b : Bool) : C := if b then s.zero else s.top
exact (C : U) (s : CostScale C) (c : C) : Bool := s.leq c s.zero

-- What the core asks of a law's value (5.S3): a verdict in `C`. For `Bool` it is `lift`; for `C`
-- the identity; for a check, `lift` of whether it holds (5.S9). A law into a type with no verdict
-- is not a law.
LawValue (C : U) (s : CostScale C) : {
  ofBool : Bool -> C,
  ofCost : C -> C,
  ofCheck : (E : U) -> Failures E -> C,
}
VerdictOfBool (C : U) (s : CostScale C) (v : LawValue C s) (b : Bool) : Bool := v.ofBool b == lift C s b
VerdictOfCost (C : U) (s : CostScale C) (v : LawValue C s) (c : C) : Bool := v.ofCost c == c
VerdictOfCheck (C : U) (s : CostScale C) (v : LawValue C s) (E : U) (fs : Failures E) : Bool :=
  v.ofCheck E fs == lift C s (isEmpty E fs)
-- the retraction: reading a Boolean verdict back gives the Boolean
ExactLift (C : U) (s : CostScale C) (v : LawValue C s) (b : Bool) : Bool := exact C s (v.ofBool b) == b
LawValue (C : U) (s : CostScale C) := VerdictOfBool C s & VerdictOfCost C s & VerdictOfCheck C s & ExactLift C s

-- ---------------------------------------------------------------- §2 The laws of relations

-- Templates a theory applies to its own relation or operation (D-2026-10-11-l): definitions the
-- laws of a theory are written with, not laws on their own, so a theory that claims them gives
-- them their subject (4.S14). The relation `r` and the operation `op` are the theory's; equality
-- is the structural `==` (4.R8) or the theory's own `eq`, passed as `same`.
isReflexive (A : U) (r : A -> A -> Bool) (x : A) : Bool := r x x
isSymmetric (A : U) (r : A -> A -> Bool) (x y : A) : Bool := not (r x y) or r y x
isAntisymmetric (A : U) (same r : A -> A -> Bool) (x y : A) : Bool := not (r x y and r y x) or same x y
isTransitive (A : U) (r : A -> A -> Bool) (x y z : A) : Bool := not (r x y and r y z) or r x z
isTotal (A : U) (r : A -> A -> Bool) (x y : A) : Bool := r x y or r y x
isAssociative (A : U) (same : A -> A -> Bool) (op : A -> A -> A) (x y z : A) : Bool := same (op (op x y) z) (op x (op y z))
isCommutative (A : U) (same : A -> A -> Bool) (op : A -> A -> A) (x y : A) : Bool := same (op x y) (op y x)
isLeftIdentity (A : U) (same : A -> A -> Bool) (op : A -> A -> A) (e x : A) : Bool := same (op e x) x
isRightIdentity (A : U) (same : A -> A -> Bool) (op : A -> A -> A) (e x : A) : Bool := same (op x e) x
isInverse (A : U) (same : A -> A -> Bool) (op : A -> A -> A) (inv : A -> A) (e x : A) : Bool :=
  same (op x (inv x)) e and same (op (inv x) x) e
isDistributive (A : U) (same : A -> A -> Bool) (times plus : A -> A -> A) (x y z : A) : Bool :=
  same (times x (plus y z)) (plus (times x y) (times x z))
isHomomorphism (A B : U) (same : B -> B -> Bool) (f : A -> B) (opA : A -> A -> A) (opB : B -> B -> B) (x y : A) : Bool :=
  same (f (opA x y)) (opB (f x) (f y))
-- The implications above (symmetric, antisymmetric, transitive) are written as alternatives:
-- sampled at random, their premises rarely hold (7.S12), so a theory quantifies them over the
-- pairs its relation relates, as kernel_laws.lth §1 does for equality.

-- ---------------------------------------------------------------- §3 A law as a theory

-- The forms of a law (D-2026-10-13-l): a case, a `forall` (Π) and an `exists` (Σ) over a type, and
-- the connectives; `then` is material, `not (p and not q)` (conditioning is an operation on a
-- measure, not a connective). This is the type `Law` of what a test runs (5.S1, 3.S12).
Law : U := data {
  Case : Bool -> Law ;
  Forall : (A : U) -> (A -> Law) -> Law ;
  Exists : (A : U) -> (A -> Law) -> Law ;
  Both : Law -> Law -> Law ;
  Either : Law -> Law -> Law ;
  Then : Law -> Law -> Law ;
}

-- Evidence follows the form of the law (D-2026-10-13-l): a case evaluated; a value of the
-- quantified type with the evidence for the law at it (a witness of an `exists`, a counterexample
-- to a `forall`); the cases listed or drawn, with their evidence; a pair for `and`; one side for
-- `or`; a citation or an assumption, which models any part whole.
Evidence : U := data {
  Evaluated : Bool -> Evidence ;
  At : (A : U) -> A -> Evidence -> Evidence ;
  Listed : List Evidence -> Evidence ;
  Drawn : Nat -> List Evidence -> Evidence ;
  Pair : Evidence -> Evidence -> Evidence ;
  Side : Bool -> Evidence -> Evidence ;
  Cited : String -> Evidence ;
  Assumed : Evidence ;
}

-- What a piece of evidence settles about a law: confirmed, refuted, supported only statistically,
-- taken on citation or assumption, or nothing. Roles: the laws of `Settling`.
Verdict : U := data {
  Confirmed : Verdict ;
  Refuted : Verdict ;
  Supported : Verdict ;
  TakenOn : String -> Verdict ;
  Unsettled : Verdict ;
}

-- A model says what evidence settles; the certifier of spec 6 and 7 is one. The laws give each
-- form of law and each kind of evidence its role.
Settling : { settles : Law -> Evidence -> Verdict }

material (a b : Bool) : Bool := not (a and not b)
both (u v : Verdict) : Verdict := match u {
  Refuted -> Refuted ;
  Confirmed -> v ;
  _ -> match v { Refuted -> Refuted ; _ -> u } }

-- a case is settled by evaluating it
CaseIsEvaluated (m : Settling) (b : Bool) : Bool :=
  m.settles (Case b) (Evaluated b) == (if b then Confirmed else Refuted)
-- (the laws below that relate a form to the evidence of its parts are stated on cases: a premise
-- on arbitrary evidence would rarely hold in a sample, 7.S12)
-- a `forall` is refuted exactly by a counterexample, and a sample over it supports and never
-- confirms (over an infinite domain only statistics confirm, 7.S17)
ForallRefutedByCounterexample (m : Settling) (A : U) (a : A) : Bool :=
  m.settles (Forall A (fn x { Case false })) (At A a (Evaluated false)) == Refuted
SampleNeverConfirms (m : Settling) (A : U) (p : A -> Law) (seed : Nat) (es : List Evidence) : Bool :=
  not (m.settles (Forall A p) (Drawn seed es) == Confirmed)
-- an `exists` is confirmed by a witness whose evidence confirms the law at it
ExistsConfirmedByWitness (m : Settling) (A : U) (a : A) : Bool :=
  m.settles (Exists A (fn x { Case true })) (At A a (Evaluated true)) == Confirmed
-- `and` by a pair, `or` by either side
BothByPair (m : Settling) (p q : Law) (e f : Evidence) : Bool :=
  m.settles (Both p q) (Pair e f) == both (m.settles p e) (m.settles q f)
EitherBySide (m : Settling) (q : Law) : Bool :=
  m.settles (Either (Case true) q) (Side true (Evaluated true)) == Confirmed
-- `then` is material
ThenIsMaterial (m : Settling) (a b : Bool) : Bool :=
  m.settles (Then (Case a) (Case b)) (Evaluated (material a b)) == (if material a b then Confirmed else Refuted)
-- a citation or an assumption models any law whole, and says on what it is taken
CitedWhole (m : Settling) (p : Law) (r : String) : Bool := m.settles p (Cited r) == TakenOn r
AssumedWhole (m : Settling) (p : Law) : Bool := m.settles p Assumed == TakenOn "assumption"

Settling := CaseIsEvaluated & ForallRefutedByCounterexample & SampleNeverConfirms & ExistsConfirmedByWitness
  & BothByPair & EitherBySide & ThenIsMaterial & CitedWhole & AssumedWhole
